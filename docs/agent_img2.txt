 1196|     text: String,
 1197|     attachments: Vec<ImageAttachment>,
 1198| ) -> Result<ChatMessage, AgentRuntimeError> {
 1199|     if attachments.len() > MAX_IMAGE_COUNT {
 1200|         return Err(AgentRuntimeError::InvalidInput(format!(
 1201|             "at most {MAX_IMAGE_COUNT} images may be attached"
 1202|         )));
 1203|     }
 1204|     let mut total = 0usize;
 1205|     let mut content = vec![ContentBlock::Text { text }];
 1206|     for attachment in attachments {

 1209|             .map_err(|_| {

 1211|         })?;

 1213|             return Err(AgentRuntimeError::InvalidInput(

 1215|             ));
 1216|         }
 1217|         total = total.saturating_add(decoded.len());
 1218|         if total > MAX_TOTAL_IMAGE_BYTES {
 1219|             return Err(AgentRuntimeError::InvalidInput(

 1221|             ));
 1222|         }
 1223|         content.push(ContentBlock::Image {
 1224|             name: attachment.name.chars().take(255).collect(),

 1226|         });
 1227|     }

 1229|         schema_version: PROTOCOL_VERSION,
 1230|         id: Uuid::new_v4().to_string(),

 1234|     })
 1235| }

 1237| fn parse_image_data_url(value: &str) -> Result<(&str, &str), AgentRuntimeError> {
 1238|     let (metadata, encoded) = value.split_once(',').ok_or_else(|| {
 1239|         AgentRuntimeError::InvalidInput("image attachment must be a data URL".into())
 1240|     })?;
 1241|     let media_type = metadata
 1242|         .strip_prefix("data:")
 1243|         .and_then(|value| value.strip_suffix(";base64"))
 1244|         .ok_or_else(|| AgentRuntimeError::InvalidInput("image data URL must use base64".into()))?;
 1245|     if !matches!(
 1246|         media_type,
 1247|         "image/png" | "image/jpeg" | "image/gif" | "image/webp"
 1248|     ) {
 1249|         return Err(AgentRuntimeError::InvalidInput(
 1250|             "image type must be PNG, JPEG, GIF, or WebP".into(),
 1251|         ));
 1252|     }
 1253|     Ok((media_type, encoded))
 1254| }

 1256| fn chat_to_provider(message: ChatMessage) -> Option<ProviderMessage> {
 1257|     let text = message.text();
 1258|     let images = message
 1259|         .content
 1260|         .into_iter()
 1261|         .filter_map(|block| match block {
 1262|             ContentBlock::Image { name, data_url } => Some(ProviderImage { name, data_url }),
 1263|             ContentBlock::Text { .. } => None,
 1264|         })
 1265|         .collect::<Vec<_>>();
 1266|     if message.role == MessageRole::User && !images.is_empty() {
 1267|         Some(ProviderMessage::UserContent { text, images })
 1268|     } else {
 1269|         Some(ProviderMessage::Text {
 1270|             role: message.role,
 1271|             text,
 1272|         })
 1273|     }
 1274| }
